# V5115: Hook threw or returned a value that is not JSON

A hook threw, or returned something the engine cannot read back as JSON: a function, a symbol, a cyclic object or
a number too large to be exact. The action it belongs to fails, and the message names the TypeScript line it threw
from. A custom check returns `true` or a failure message rather than throwing, so a failed check reads as a check
result, not as a broken hook.

## Wrong

```yaml
# file: services/orders/orders.test.yaml
tests:
  - id: listsSortedOrders
    call: orders#listOrders
    expect:
      custom:
        - function: ../../hooks/checks.ts#isSorted
```

```ts
# file: hooks/checks.ts
export function isSorted({ response }: { response: { body: number[] } }): boolean {
  if (response.body.some((n, i) => i > 0 && response.body[i - 1] > n)) {
    throw new Error("not sorted");
  }
  return true;
}
```

## Right

```yaml
# file: services/orders/orders.test.yaml
tests:
  - id: listsSortedOrders
    call: orders#listOrders
    expect:
      custom:
        - function: ../../hooks/checks.ts#isSorted
```

```ts
# file: hooks/checks.ts
export function isSorted({ response }: { response: { body: number[] } }): boolean | string {
  const sorted = response.body.every((n, i) => i === 0 || response.body[i - 1] <= n);
  return sorted || "the orders are not sorted";
}
```
